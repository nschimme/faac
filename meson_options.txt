option('frontend',
    description: 'Build the frontend',
    type: 'boolean',
    value: true)

# SBR spectral analysis density: analyse every Nth slot per frame.
# 1 = full quality (every slot); higher values trade MOS for faster encode.
# Max 8: above that, both halves of the frame are sampled but the gain is negligible.
# Only affects HE-AAC; LC encodes are identical regardless of this setting.
#   1: every slot  -- reference quality (MOS d: 0)
#   2: every 2nd   -- ~20% faster HE-AAC encode; MOS d: -0.004
#   4: every 4th   -- ~30% faster HE-AAC encode; MOS d: -0.015
#   8: every 8th   -- ~36% faster HE-AAC encode; MOS d: -0.028
#
# HE-AAC v2 (parametric stereo) pays a second, separate cost here that HE-AAC v1
# does not: its IID/ICC parameters are estimated from the same decimated slots,
# so decimating raises the estimator's variance and blurs the stereo image.
# Measured as inter-channel coherence error (12 clips, 16 kbps stereo); HE-AAC v1
# is flat across the same sweep, so this column is PS-specific:
#   1: coherence err 0.0932 (reference)   HE-v2 encode speed: +0%
#   2: coherence err 0.0938 (+0.6%)       HE-v2 encode speed: +23%
#   4: coherence err 0.0981 (+5.3%)       HE-v2 encode speed: +33%
#   8: coherence err 0.1077 (+15.5%)      HE-v2 encode speed: +40%
# N=2 is close to free for PS; N=8 costs a sixth of the image accuracy PS exists
# to provide. Prefer <=4 when encoding HE-AAC v2.
option('sbr-decimation',
    description: 'SBR spectral analysis density: analyse every Nth slot (1=full quality, max=8)',
    type: 'integer',
    min: 1,
    max: 8,
    value: 1)

option('max-channels',
    description: 'Maximum number of channels',
    type: 'integer',
    min: 1,
    max: 8,
    value: 8)

# HE-AAC v2 parametric stereo (ISO/IEC 14496-3 §8.6.4).
# When false, HE-AAC v2 is compiled out entirely: --object-type he-aac-v2 is
# rejected at configuration time and the encoder is byte-identical to a build
# that never had the feature. LC and HE-AAC v1 output is identical either way.
# Costs ~4.5% library size when enabled.
option('parametric-stereo',
    description: 'Support HE-AAC v2 (parametric stereo); disable to compile it out entirely',
    type: 'boolean',
    value: true)
