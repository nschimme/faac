option('frontend',
    description: 'Build the frontend',
    type: 'boolean',
    value: true)

option('floating-point',
    description: 'Floating point precision',
    type: 'combo',
    choices: ['single', 'double'],
    value: 'double')

option('max-channels',
    description: 'Maximum number of channels',
    type: 'integer',
    min: 1,
    max: 8,
    value: 8)

# FFT implementation type: pick the algorithm for spectral analysis.
# radix4: Radix-4 Decimation-in-Frequency (DIF) (optimized).
# radix2: Standard Radix-2 implementation (portable, baseline speed).
#   radix4: ~10% faster overall encode; MOS d: 0 -- default
#   radix2: reference implementation (MOS d: 0)
option('fft-type',
    description: 'FFT implementation type (radix2: standard, radix4: optimized DIF)',
    type: 'combo',
    choices: ['radix2', 'radix4'],
    value: 'radix4')
